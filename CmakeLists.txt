cmake_minimum_required(VERSION 3.16)

project(threadpool_mempool LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_executable(tpmp
    src/main.cpp
    src/ThreadPool.cpp
    src/MemoryPool.cpp
)

target_include_directories(tpmp PRIVATE include)
